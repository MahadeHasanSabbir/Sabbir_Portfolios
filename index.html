aminship / AreaCalculator.php 
public function calculate($corners) {
  $area = 0;
  foreach ($corners as $i => $p) {
    $next = $corners[($i+1) % count($corners)];
    $area += $p->x * $next->y
           - $next->x * $p->y;
  }
  return abs($area) / 2; // shoelace
}

inventory / StockController.php 
public function checkout(Order $order) {
  foreach ($order->items as $item) {
    $product = $item->product;
    if ($product->stock < $item->qty)
      throw new OutOfStockException($product);

    $product->decrement('stock', $item->qty);
  }
  LowStockAlert::dispatchIf($product->stock < 10);
}
